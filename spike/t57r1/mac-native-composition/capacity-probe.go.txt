package main
import("context";"encoding/json";"fmt";"os";"runtime";"time";"github.com/bmeddeb/phebs/internal/lifecycle")
func main(){ctx,cancel:=context.WithTimeout(context.Background(),5*time.Second);defer cancel();ok:=true
for _,path:=range os.Args[1:]{c,e:=lifecycle.NewGate(path).Check(ctx,0);errText:="";if e!=nil{errText=e.Error()};ready:=e==nil&&c.Pressure==lifecycle.PressureNormal&&c.UsedPercent<lifecycle.ResumeWatermarkPercent
v:=struct{Path string `json:"path"`;GOOS string `json:"goos"`;GOARCH string `json:"goarch"`;Capacity lifecycle.Capacity `json:"capacity"`;Error string `json:"error"`;BelowResume bool `json:"below_resume"`;Ready bool `json:"ready"`}{path,runtime.GOOS,runtime.GOARCH,c,errText,c.UsedPercent<lifecycle.ResumeWatermarkPercent,ready};if err:=json.NewEncoder(os.Stdout).Encode(v);err!=nil{fmt.Fprintln(os.Stderr,err);os.Exit(1)};ok=ok&&ready};if !ok{os.Exit(1)}}
